--------------------------------------------------------------------------------
3. USEFUL QUERIES: PUBLIC SCHEMA (BUSINESS DATA)
--------------------------------------------------------------------------------

-- 1. View Destination Weddings (Top 10)
SELECT code, title, venue, city, status, guest_count_planned, 
       (budget_paise / 1000000000.0)::numeric(10,2) || ' Cr' AS budget_inr
FROM public.weddings
ORDER BY start_date DESC
LIMIT 10;

-- 2. View Sales Enquiries Pipeline (Top 15)
SELECT contact_name, city, estimated_guests, status, created_at::date AS enquiry_date
FROM public.enquiries
ORDER BY created_at DESC
LIMIT 15;

-- 3. View Enquiries Count by Status
SELECT status, count(*) AS total_enquiries
FROM public.enquiries
GROUP BY status
ORDER BY total_enquiries DESC;

-- 4. View Curated Vendors by Category
SELECT name, category, city, rating, base_price_paise / 100 AS base_price_inr
FROM public.vendors
ORDER BY category, rating DESC
LIMIT 15;

-- 5. View Operational Tasks for a Wedding
SELECT t.title, t.category, t.status, t.due_date, w.title AS wedding
FROM public.tasks t
JOIN public.weddings w ON w.id = t.wedding_id
LIMIT 15;

-- 6. View Vendor Bills and Approved Amounts
SELECT b.bill_number, v.name AS vendor_name, b.status, 
       (b.amount_paise / 100.0)::numeric(12,2) AS bill_amount_inr,
       (b.paid_amount_paise / 100.0)::numeric(12,2) AS paid_amount_inr
FROM public.vendor_bills b
JOIN public.vendors v ON v.id = b.vendor_id
ORDER BY b.created_at DESC
LIMIT 15;

-- 7. View Client Advance Receipts
SELECT receipt_number, (amount_paise / 100.0)::numeric(12,2) AS amount_inr, 
       payment_method, reference, received_at::date
FROM public.client_receipts
ORDER BY received_at DESC
LIMIT 10;

-- 8. View User Profiles by Role (Admin, Employee, Client)
SELECT full_name, email, role, phone, created_at::date
FROM public.profiles
ORDER BY role, created_at DESC
LIMIT 20;


--------------------------------------------------------------------------------
4. USEFUL QUERIES: PRIVATE SCHEMA (SECURITY & AUTH DATA)
--------------------------------------------------------------------------------

-- 1. View Password Hashes (Argon2id - notice passwords are never plaintext)
SELECT profile_id, 
       LEFT(password_hash, 35) || '...' AS argon2id_hash, 
       failure_count, 
       locked_until, 
       updated_at
FROM private.auth_credentials
LIMIT 10;

-- 2. View Active Logged-In Sessions
SELECT s.id, p.email, p.role, 
       LEFT(s.token_hash, 16) || '...' AS session_token_hash, 
       s.expires_at, 
       s.last_seen_at
FROM private.sessions s
JOIN public.profiles p ON p.id = s.profile_id
ORDER BY s.last_seen_at DESC
LIMIT 10;

-- 3. View Google OAuth Linked Accounts
SELECT o.email, o.provider, o.provider_sub, p.role, o.created_at
FROM private.oauth_identities o
JOIN public.profiles p ON p.id = o.profile_id;

-- 4. View Security Login Attempts & Lockout Audit Trail
SELECT email, ip_address, success, attempted_at
FROM private.login_attempts
ORDER BY attempted_at DESC
LIMIT 15;

-- 5. View Active OAuth PKCE Transactions
SELECT id, state_hash, return_to, expires_at, created_at
FROM private.oauth_transactions
ORDER BY created_at DESC
LIMIT 5;

-- 6. View Employee Invitation Tokens
SELECT email, role, full_name, hourly_rate_paise / 100 AS hourly_rate_inr, 
       expires_at, used_at
FROM private.invite_tokens
ORDER BY created_at DESC;
